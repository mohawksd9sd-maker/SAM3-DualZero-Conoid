\begin{abstract}
We study the right conoid immersed by
$r(u,v)=(u\cos v,\,u\sin v,\,\ell_0\sin(2v))$, with the induced metric
$ds^2=du^2+f^2\,dv^2$ and $f=\sqrt{u^2+4\ell_0^2\cos^2(2v)}$.
We prove the induced-metric formulae, the exact $C_4$ isometry
(and that $A_5$ is not an isometry of $f$), the Gaussian curvature,
the evaluation $C_0=2\pi\ln 2$ of the regularised volume finite part,
a strict variational monotonicity of radial channel ground energies in
the angular momentum $m$ (with a domain caveat at $m=0$), the exact
formula $E_0(m;U)=(j_{\nu,1}/U)^2$, and that the APS index of the
untwisted Dirac operator on the truncated conoid is $0$ for the
bounding spin structure (Theorem~\ref{thm:aps0}).
On the unit metric $\ell_0=1$ we prove $C_0=2\pi\ln 2$; an earlier draft band $4.43$--$4.45$ does not reproduce.
Claims are labelled as theorem, proposition, numerical observation,
heuristic, convention, or conjecture. We do not claim a derivation of
the full Standard Model, a scheme-free formula for $G_N$, a derivation
of $\omega_0\approx 0.927$ or of the exponent $4/13$, an APS index of
three for the untwisted truncated conoid (that index is $0$), or a
proof of the Riemann hypothesis.
\end{abstract}

\section{Geometry}

\begin{definition}[Locked embedding]
\label{def:embedding}
For $\ell_0>0$ and $(u,v)\in[0,\infty)\times\mathbb{R}/2\pi\mathbb{Z}$,
\[
  r(u,v)=\bigl(u\cos v,\; u\sin v,\; \ell_0\sin(2v)\bigr)\in\mathbb{R}^3.
\]
\end{definition}

\begin{proposition}[Induced metric]
\label{prop:metric}
The induced Riemannian metric is
\begin{equation}
  ds^2=du^2+f(u,v)^2\,dv^2,
  \qquad
  f=\sqrt{u^2+4\ell_0^2\cos^2(2v)}.
\end{equation}
In particular $g_{uu}=1$, $g_{uv}=0$, and the coefficient of
$\ell_0^2\cos^2(2v)$ in $g_{vv}$ is $4$, not $16$.
\end{proposition}

\begin{proof}
Differentiating the embedding gives
$r_u=(\cos v,\sin v,0)$ and
$r_v=(-u\sin v,\,u\cos v,\,2\ell_0\cos(2v))$.
Hence $g_{uu}=r_u\cdot r_u=1$, $g_{uv}=r_u\cdot r_v=0$, and
$g_{vv}=u^2+4\ell_0^2\cos^2(2v)$.
\end{proof}

\begin{proposition}[$C_4$ isometry; $A_5$ is not an isometry]
\label{prop:c4}
The map $v\mapsto v+\pi/2$ is an isometry of $ds^2$. A rotation of $v$
by $2\pi/5$ is not: already $f(u,v)\neq f(u,v+2\pi/5)$ at
$(u,v,\ell_0)=(0.3,0.2,1)$. Consequently $A_5$ is not a continuous
isometry group of the locked metric.
\end{proposition}

\begin{proof}
$\cos(2(v+\pi/2))=\cos(2v+\pi)=-\cos(2v)$, so $f$ is invariant, and the
coordinate form of $ds^2$ is unchanged. The $2\pi/5$ counterexample is
a direct evaluation.
\end{proof}

\begin{proposition}[Tip nodes]
\label{prop:tips}
On $u=0$, $f(0,v)=0$ if and only if $\cos(2v)=0$, i.e.\ at the four
points $v=\pi/4+k\pi/2$ for $k=0,1,2,3$.
\end{proposition}

\begin{proposition}[Gaussian curvature]
\label{prop:curvature}
Away from $\{f=0\}$, the Gaussian curvature of $ds^2=du^2+f^2\,dv^2$ is
\[
  K=-\frac{f_{uu}}{f}=-\frac{4\ell_0^2\cos^2(2v)}{f^4}.
\]
In particular, at $u=0$ and $|\cos(2v)|=1$ one has
$|K|_{\max}=1/(4\ell_0^2)$.
\end{proposition}

\begin{proof}
With $v$ held fixed, $f^2=u^2+4\ell_0^2\cos^2(2v)$, so
$f_u=u/f$ and $f_{uu}=4\ell_0^2\cos^2(2v)/f^3$. For an orthogonal
metric of this form, $K=-f_{uu}/f$, which is the displayed formula.
\end{proof}

\section{Regularised volume}

The area of the truncated surface $0\le u\le U$ is
$\mathrm{Vol}(U)=\int_0^U\int_0^{2\pi} f\,dv\,du$.
Write $I(u)=\int_0^{2\pi}(f-u)\,dv$, so that
$\mathrm{Vol}(U)=\pi U^2+\int_0^U I(u)\,du$.

\begin{proposition}[Existence of $C_0$]
\label{prop:c0exists}
Let $\ell_0=1$. Then $I(u)=2\pi/u+O(u^{-3})$ as $u\to\infty$, and the
limit
\begin{equation}
  C_0=\lim_{U\to\infty}\Bigl(\int_0^U I(u)\,du-2\pi\log U\Bigr)
\end{equation}
exists and is finite. The same limit is obtained if $\log U$ is replaced
by $\log(U+1)$.
\end{proposition}

\begin{proof}
For $u>0$,
$f-u=u\bigl(\sqrt{1+4\cos^2(2v)/u^2}-1\bigr)
=2\cos^2(2v)/u+O(u^{-3})$,
uniformly in $v$. Integrating in $v$ and using
$\int_0^{2\pi}\cos^2(2v)\,dv=\pi$ gives $I(u)=2\pi/u+O(u^{-3})$.
Thus $\int_1^\infty(I(u)-2\pi/u)\,du$ converges absolutely, and $C_0$
exists. Finally $\log(U+1)-\log U\to 0$.
\end{proof}

Hence
$\mathrm{Vol}(U)=\pi U^2+2\pi\log U+C_0+o(1)$.

\begin{theorem}[$C_0=2\pi\ln 2$]
\label{thm:c0}
On the unit metric $\ell_0=1$, the finite part of Proposition~\ref{prop:c0exists}
is exactly $C_0=2\pi\ln 2$.
\end{theorem}

\begin{proof}
Write $a(v)=2\lvert\cos(2v)\rvert$, so $f=\sqrt{u^2+a(v)^2}$. For each fixed
$a>0$,
\[
  \int_0^U\bigl(\sqrt{u^2+a^2}-u\bigr)\,du
  =\frac{a^2}{4}+\frac{a^2}{2}\ln\frac{2U}{a}+o(1).
\]
Integrating in $v\in[0,2\pi]$ and subtracting $2\pi\log U$ therefore yields
\begin{align*}
  C_0
  &=\pi+2\pi\ln 2-\frac12\int_0^{2\pi}a(v)^2\ln a(v)\,dv\\
  &=\pi-8K,
\end{align*}
where the elementary identities $\int_0^{2\pi}\cos^2(2v)\,dv=\pi$ and
$\int a^2\ln a=2\pi\ln 2+8K$ have been used, and
\[
  K=\int_0^{\pi/2}\cos^2 t\,\ln(\cos t)\,dt.
\]
The classical Fourier series
$\ln(\cos t)=-\ln 2+\sum_{k=1}^\infty(-1)^{k+1}k^{-1}\cos(2kt)$,
valid for $\lvert t\rvert<\pi/2$
(equivalent to $-\ln(\cos t)=\ln 2+\sum_{k=1}^\infty(-1)^k k^{-1}\cos(2kt)$),
gives, after integrating against $(1+\cos 2t)/2$,
$K=\pi(1-2\ln 2)/8$. Hence $C_0=\pi-8K=2\pi\ln 2$.
\end{proof}

\begin{numobs}
Quadrature of the original integral is consistent with the theorem:
$C_0(U=1000)-2\pi\ln 2\approx 2.36\times 10^{-6}$ and
$(C_0(U)-2\pi\ln 2)\,U^2$ is approximately constant, as expected from the
$O(U^{-2})$ remainder. The earlier draft band $4.43$--$4.45$ does not
reproduce and is withdrawn.
\end{numobs}

\section{Self-adjointness of radial channels}

On $(0,U)$ consider the radial operators
$H_m=-\partial_{uu}+m^2(u+\varepsilon)^{-2}+V_{\mathrm{tip}}(u)$
with $\varepsilon\ge 0$ a possible cutoff and $V_{\mathrm{tip}}$ a
real, relatively bounded tip potential (including the case $V_{\mathrm{tip}}=0$).

\begin{proposition}[Weyl classification]
\label{prop:weyl}
For the inverse-square potential $\kappa/u^2$ on $(0,1)$, the operator
$-d^2/du^2+\kappa/u^2$ is limit-circle at $u=0$ if $\kappa<3/4$ and
limit-point if $\kappa\ge 3/4$. Consequently, for $\kappa=m^2$, the
channel $m=0$ is limit-circle at the tip, while every channel with
$|m|\ge 1$ is limit-point (unique self-adjoint extension).
\end{proposition}

This is the standard Weyl criterion for inverse-square potentials; see
Reed--Simon~\cite{ReedSimon}. For $m=0$ a self-adjoint realization must
be chosen. In the numerical checks below we use a Dirichlet cutoff on a
compact interval, which selects one such realization; an APS/conic tip
condition would be another. Variational comparisons that include $m=0$
are understood on a fixed common realization.

\section{Channel monotonicity}

\begin{theorem}[Strict monotonicity]
\label{thm:mono}
Let $m_2>m_1\ge 1$, and let $H_{m_i}$ be the unique self-adjoint
realizations on a finite interval $(0,U)$ (or on $(0,\infty)$ with a
confining condition at infinity). Then the ground energies satisfy
$E_0(m_2)>E_0(m_1)$. The same strict inequality holds for $m_1=0$ and
$m_2\ge 1$ on any common form domain in which the quadratic form of
$H_0$ is closed and $\int(u+\varepsilon)^{-2}|\psi|^2$ is finite and
not identically zero for nonzero form-domain functions (in particular,
on a Dirichlet cutoff).
\end{theorem}

\begin{proof}
On the indicated form domain,
$H_{m_2}=H_{m_1}+(m_2^2-m_1^2)(u+\varepsilon)^{-2}$
as quadratic forms, and the added potential is nonnegative and not
almost-everywhere zero. Hence $E_0(m_2)\ge E_0(m_1)$. If equality held,
a ground state $\varphi$ of $H_{m_2}$ would satisfy
$\langle H_{m_2}\rangle_\varphi=E_0(m_1)$, while
\[
  \langle H_{m_2}\rangle_\varphi
  =\langle H_{m_1}\rangle_\varphi+(m_2^2-m_1^2)\int(u+\varepsilon)^{-2}|\varphi|^2
  \ge E_0(m_1)+(m_2^2-m_1^2)\int(u+\varepsilon)^{-2}|\varphi|^2.
\]
The integral is strictly positive for $\varphi\not=0$, a contradiction.
(The one-line comparison of $H_{m_2}$ against a ground state of $H_{m_1}$
alone yields only $E_0(m_2)\le\langle H_{m_2}\rangle_{\psi_{m_1}}$, which
is the wrong direction; the form inequality is what is needed.)
\end{proof}

\begin{corollary}
\label{cor:e2e3}
$E_0(2)<E_0(3)$ on these realizations. Any spectral threshold lying
strictly between them isolates the three channels $m\in\{0,1,2\}$ among
the radial model operators $H_m$.
\end{corollary}

\begin{theorem}[Exact ground energies of the radial model]
\label{thm:E0exact}
Let $H_m=-\frac{d^2}{du^2}+m^2/u^2$ act on $(0,U)$ with the regular
solution at $u=0$ (the unique self-adjoint extension for
$\lvert m\rvert\ge 1$; the regular choice for $m=0$) and Dirichlet
data at $u=U$. Then
\[
  E_0(m;U)=\Bigl(\frac{j_{\nu,1}}{U}\Bigr)^2,
  \qquad \nu=\sqrt{m^2+\tfrac14},
\]
where $j_{\nu,1}$ is the first positive zero of the Bessel function
$J_\nu$. In particular $j_{1/2,1}=\pi$, so $E_0(0;U)=\pi^2/U^2$.
The ratios $E_0(m)/E_0(n)$ are independent of $U$.
\end{theorem}

\begin{proof}
The substitution $\psi=\sqrt{u}\,y$ reduces $-\psi''+(m^2/u^2)\psi=E\psi$
to Bessel's equation of order $\nu=\sqrt{m^2+1/4}$ in the variable
$z=\sqrt{E}\,u$. The solution regular at the origin is $y(z)=J_\nu(z)$
(the companion $Y_\nu$ is not square-integrable at $0$ in the relevant
range). The condition $\psi(U)=0$ forces $J_\nu(\sqrt{E}\,U)=0$, hence
the stated eigenvalues; the ground state is the first positive zero.
For $m=0$ one has $J_{1/2}(z)=\sqrt{2/(\pi z)}\sin z$, whose first
positive zero is $\pi$.
\end{proof}

The same operator also satisfies the elementary bounds
$E_0(m)\ge\pi^2/U^2$ (Theorem~\ref{thm:mono}) and
$E_0(m)>(m^2+1/4)/U^2$ (since $j_{\nu,1}>\nu$ for $\nu\ge 0$), together
with the variational upper bound obtained from the trial function
$\sin(\pi u/U)$,
\[
  E_0(m)\le \frac{\pi^2+2\pi m^2\int_0^\pi(\sin t/t)^2\,dt}{U^2},
\]
with equality at $m=0$.

\begin{numobs}
The exact values at $U=20$ are
$E_0(0)=\pi^2/400\approx 0.02467401$,
$E_0(1)\approx 0.03980128$,
$E_0(2)\approx 0.06795432$,
$E_0(3)\approx 0.10338722$.
A Dirichlet finite-difference scheme with $800$ interior points agrees
to a relative error of about $10^{-6}$. These are eigenvalues of the
radial model operators, not of the $2$D Dirac operator on the conoid.
\end{numobs}

\begin{remark}[Soft set, heuristic]
The metric factor $\cos(2v)$ is the $m=2$ angular harmonic, and the
exact $C_4$ isometry is compatible with a consecutive block
$\{0,1,2\}$. This is a preference heuristic for calling $\{0,1,2\}$ the
soft set of the radial model. It is \emph{not} a uniqueness theorem, and
it is \emph{not} a computation of the APS index, or of the number of
light eigenvalues, of the two-dimensional Dirac operator on the conoid.
\end{remark}

\begin{remark}[Withdrawn as a truncated-untwisted conjecture]
\label{conj:three}
An earlier draft conjectured that, under a specified APS-compatible
spin structure on a suitable truncation of the conoid, the Dirac
operator would have exactly three light channels or index three.
For the \emph{untwisted} Dirac operator on the truncated conoid
$X_{a,b}$, with the bounding spin structure, that claim is false:
Theorem~\ref{thm:aps0} gives index $0$. The singular-tip limit
$a\to 0$ is not covered by that theorem and remains open; it is not
an index-$3$ result. The numerical Dirac solver of
Section~\ref{sec:dirac} is a truncation, not a substitute for the
theorem. The label \texttt{conj:three} is retained only for
cross-references.
\end{remark}

\section{Spectral action and $G_N$}

On a product spectral triple, the Seeley--DeWitt coefficient $a_2$ of
the spectral action~\cite{ChamseddineConnes} produces an
Einstein--Hilbert term whose coupling involves the internal scale
$\ell_0$. This yields the structural relation $G_N=\alpha\ell_0^2$.

The pure prefactor $\alpha$ depends on normalisation conventions
(cutoff moments, the internal volume regularisation, and any overall
factor in the spectral action). The repository records the normalisation
$G_N=64\pi\ell_0^2/45$, which is equivalent to fixing a single
combination of those conventions
(cf.\ the discussion in the project's mathematical notes). This is a
\emph{convention}, not a scheme-free derivation. The proportionality
$G_N\propto\ell_0^2$ is the structural statement; the number $64\pi/45$
is not independently derived here.

\section{$\omega_0$ and the exponent $4/13$}
\label{sec:omega}

The Dual-Zero generator is a definition,
$\varepsilon(n)=\omega_0(-1)^n n^{-n}$, not a theorem. The value
$\omega_0\approx 0.927$ and the formula
$\omega_0=(R_{\mathrm{curv}}/D_{\mathrm{bridge}})^{4/13}$ with
$D_{\mathrm{bridge}}=2\pi/12$ are \emph{not derived} in this note.

What is missing is: (i)~a unique geometric definition of the curvature
radius $R_{\mathrm{curv}}$; (ii)~a derivation of the exponent $4/13$ from
the metric, the spectral action, or a closed variational problem;
(iii)~a reason to pair that $R$ with $D_{\mathrm{bridge}}=2\pi/12$.

Even granting the exponent, natural readings of $R$ at the curvature
maximum ($u=0$, $|\cos(2v)|=1$) do not yield $0.927$ at $\ell_0=1$:
\[
\begin{array}{lcc}
\hline
R & R/D & (R/D)^{4/13}\\
\hline
1/|K|_{\max}=4\ell_0 & 7.639\ldots & 1.869\ldots\\
1/\sqrt{|K|_{\max}}=2\ell_0 & 3.820\ldots & 1.510\ldots\\
\ell_0 & 1.910\ldots & 1.220\ldots\\
\hline
\end{array}
\]
The value $0.927$ would require $R\approx 0.409\,\ell_0$, which is not
one of these geometric prescriptions. Treating $0.927$ as an input
parameter is honest; treating it as a derived output is not.

\section{Numerical Dirac operator}
\label{sec:dirac}

A previous real-matrix assembly followed by symmetrisation
$D\mapsto\tfrac12(D+D^T)$ is not a Dirac operator: on a flat torus the
derivative terms cancel and the spectrum is not the analytic Dirac
spectrum. A complex Hermitian discretisation of
\[
  D=i\sigma^1\bigl(\partial_u+\tfrac12\partial_u\log f\bigr)+i\sigma^2 f^{-1}\partial_v
\]
is implemented in \texttt{code/aps\_hermitian.py}.

\begin{numobs}[Flat torus, known answer]
On the flat torus $\mathbb{R}^2/(2\pi\mathbb{Z})^2$ with antiperiodic
conditions in both directions, the analytic spectrum is
$\pm\sqrt{k_u^2+k_v^2}$ with $k=n+\tfrac12$, $n\in\mathbb{Z}$. The
lowest $\lvert\lambda\rvert$ is $\sqrt{1/2}$. At grid size $N=24$ the
Hermitian scheme matches the eight copies of $\pm\sqrt{1/2}$ with
maximum absolute error $\approx 0.00202$. A log--log fit of the error
on $N\in\{8,12,16,24,32\}$ has slope $\approx-1.995$, consistent with
second-order central differences. The operator is Hermitian to machine
precision.
\end{numobs}

On the truncated conoid, a boundary-penalty discretisation of APS was
also implemented. Its eigenvalues are a numerical experiment and are
not used as a substitute for Theorem~\ref{thm:aps0} below.

\section{APS index and the generation count}
\label{sec:aps}

The APS boundary condition is the spectral projection onto the nonnegative
eigenspace of the induced boundary operator~\cite{APS1975,FalomirDisk}.
For the free Dirac operator on a flat disk of radius $R$ with flux
$\kappa=0$, there are no zero modes and the nonzero eigenvalues are
$\pm j_{n,l}/R$, the zeros of the Bessel functions $J_n$; for
$k<\kappa\le k+1$ the index is $k+1$~\cite{FalomirDisk}.

A mode-by-mode radial discretisation with SAT enforcement of the Falomir
conditions reproduces some of this Bessel spectrum (the $n=1$ family to
about $0.02$ at $N=80$), but the operator is not exactly Hermitian and a
numerical reproduction of the \emph{nonzero}-flux zero-mode count was not
obtained. Those failures are reported as such. The continuum statement
that follows does not depend on that discretisation.

\begin{theorem}[Untwisted APS index on the truncated conoid]
\label{thm:aps0}
Let $X_{a,b}=[a,b]\times S^1$ with $0<a<b<\infty$ carry the locked
right-conoid metric, and let $D$ be the untwisted $2$D Dirac operator.
After the product-collar deformation required by the Atiyah--Patodi--Singer
theorem, the bulk $\hat A$-density vanishes in dimension $2$, and each
boundary operator is unitarily equivalent to the standard one-dimensional
Dirac operator on a circle of length $L(u)=\oint f(u,\cdot)\,dv$. For the
antiperiodic (bounding) spin structure on each end --- the structure that
extends over a disk cap --- one has $\eta=h=0$, hence $\bar\eta=0$, at
each boundary component. Therefore
\[
  \operatorname{ind} D_{\mathrm{APS}}(X_{a,b})=0.
\]
The value is independent of $a$, $b$, and $\ell_0$. In particular it is
not equal to $3$.
\end{theorem}

A periodic spin structure on a boundary component contributes
$\bar\eta=1/2$. In every standard orientation convention the resulting
index then lies in $\{-1,0,1\}$, never $3$.

The theorem applies to the truncated conoid as a smooth compact manifold
with boundary. The singular-tip limit $a\to 0$ is not covered: the
metric fails to be smooth at the four tip nodes, and APS as stated
requires a smooth compact manifold with boundary.

A nonzero index can be arranged by coupling to an auxiliary $U(1)$
connection with boundary holonomy $\kappa$. By the Falomir--Pisani--Wipf
disk calculation, index $3$ requires $\kappa\in(2,3]$. That flux is not
determined by the locked metric: the spin connection of $f$ is already
included in the untwisted Dirac operator and must not be counted twice.

\begin{numobs}[Three light channels]
Conjecture~\ref{conj:three}, read as the claim that the APS index of the
untwisted Dirac operator on the truncated conoid equals $3$, is
\emph{refuted} by Theorem~\ref{thm:aps0}. The singular-tip problem
remains open and is not an index-$3$ theorem. The radial-model heuristic
$\{0,1,2\}$ is unchanged and is not an index statement. A Falomir flux
$\kappa\in(2,3]$ would give index $3$ on a disk, but is an extra input,
not a derivation.
\end{numobs}

\section{$\omega_0$ and $4/13$, continued}

The search recorded in \texttt{code/omega0\_search.py} found no unique
geometric $R_{\mathrm{curv}}$ and no derivation of the exponent $4/13$.
The exponents $p$ that would be needed to obtain $0.927=(R/D)^p$ from the
natural radii of Section~\ref{sec:omega} are all negative (since those
$R/D>1$), hence far from $4/13$. The closest simple constants,
$\sqrt{6/7}\approx 0.9258$ and $13/14\approx 0.9286$, differ from $0.927$
by more than $10^{-3}$ and are not derivations. Dual-Zero weights are
linear in $\omega_0$. The parameter $\omega_0$ remains a free input.

\section{Scheme table for the prefactor $\alpha$}
\label{sec:alpha}

Under the structural identification of notes~09 and~14,
$\alpha=1/(16\pi\,C_{\mathrm{SA}}\,\phi_2\,\mathrm{Vol}_{\mathrm{reg}})$
when $\Lambda=1/\ell_0$ and $G_N=\kappa^2/(8\pi)$. For
$\mathrm{Vol}_{\mathrm{reg}}=C_0=2\pi\ln 2$ and $\phi_2=\tfrac12$
(sharp cutoff or $f(v)=e^{-v^2}$), the hypothesis
$C_{\mathrm{SA}}=1/(24\pi^2)$ gives $\alpha\approx 2.164$, about $0.484$
times $64\pi/45$. The hypothesis $C_{\mathrm{SA}}=1/(4\pi^2)$ gives
$\alpha\approx 0.361$. Using a truncated area $\pi U^2+2\pi\log U+C_0$
changes $\alpha$ by a large factor. The recorded $64\pi/45$ remains a
convention lock, equivalent to
$C_{\mathrm{SA}}\phi_2\mathrm{Vol}_{\mathrm{reg}}=45/(1024\pi^2)$.
No unique geometric $\alpha$ is claimed. The Chamseddine--Connes--Marcolli
Einstein--Hilbert coefficient for the Standard Model almost-commutative
geometry includes a finite Hilbert space that is not the conoid, and is
not used here to produce an $\alpha$.

\section{Limitations}

\begin{enumerate}
\item \emph{The singular tip is not treated.}
  Theorem~\ref{thm:aps0} gives the APS index of the truncated
  (smooth) conoid, which is $0$ for the bounding spin structure.
  Edge and $b$-calculus are cited~\cite{APS1975,MelroseAPS}, not applied
  to a complete parametrix at the four singular tip nodes ($a\to 0$).
  Conjecture~\ref{conj:three} is refuted in the truncated-untwisted
  reading and remains open only as a singular-tip problem, without
  an index-$3$ theorem.
\item \emph{The $m=0$ channel is limit-circle.} Monotonicity including
  $m=0$ depends on a chosen self-adjoint realization.
\item \emph{The soft set is a radial-model heuristic}, not a count of
  Dirac eigenvalues.
\item \emph{The prefactor $64\pi/45$ is a convention lock.}
\item \emph{$\omega_0$ and $4/13$ are not derived}
  (Section~\ref{sec:omega}).
\item \emph{Lorentzian signature is not treated.}
\item \emph{Yukawa eigenvalues, CKM and PMNS angles, a Higgs mass, gauge
  unification, and the cosmological constant are not derived here.}
  Scripts in the repository that appear to ``reproduce'' such numbers
  largely echo hardcoded constants or rearrange definitions; they are
  not evidence.
\item \emph{A twelve-bridge $A_5$ module}, if used at all, is a discrete
  labelling device inside the soft sector. It is not an isometry group
  of $f$, and it is not used in any theorem of this note.
\item \emph{The Dual-Zero generator is a definition.} No identification
  of its critical points with the zeros of the Riemann zeta function is
  claimed; a non-circular test in the repository found no such evidence.
\end{enumerate}

\section{Non-claims}

We do not claim a full Standard Model derivation, percent-level
unification, resolution of the cosmological constant, UV-complete
quantum gravity, uniqueness of the conoid, or a proof of the Riemann
hypothesis. We do not claim that $A_5$ is a continuous isometry group of
the locked metric, nor that the $C_4$ isometry group determines
$\mathrm{SU}(3)\times\mathrm{SU}(2)\times\mathrm{U}(1)$. We do not claim
that $G_N=64\pi\ell_0^2/45$ or
$\omega_0=(R/D)^{4/13}\approx 0.927$ has been derived from the geometry
alone, nor that the exact ratios $E_0(m)/E_0(n)$ are Standard Model mass
ratios, nor that gauge couplings unify at a derived scale. We do not
claim an APS index of $3$ for the untwisted Dirac operator on the
truncated conoid; that index is $0$ (Theorem~\ref{thm:aps0}).
