\section{Dual-Zero Hyperreal Regulator}
\label{sec:regulator}

\begin{definition}
Fix \(\omega_0 > 0\). Define \(\varepsilon(n) = \omega_0 (-1)^n n^{-n}\). Let \(\mathcal{U}\) be a non-principal ultrafilter on \(\mathbb{N}\). The Dual-Zero regulator is
\[
\mathrm{Reg}_2(\varepsilon) := \mathrm{st}_{\mathcal{U}}\Bigl( \frac{\varepsilon(2n) + \varepsilon(2n+1)}{2} \Bigr).
\]
It acts on functions/measures via \(\mathrm{Reg}_2(f)(n) := \frac{f(2n) + f(2n+1)}{2}\).
\end{definition}

\begin{axiom}[Linearity -- A1]
\(\mathrm{Reg}_2\) extends linearly to finite signed measures on \([0,\infty)\).
\end{axiom}

\begin{axiom}[Grading Compatibility -- A2]
Antisymmetric signed measures are annihilated: \(\mathrm{Reg}_2(\nu) = 0\) if \(\nu \mapsto -\nu\).
\end{axiom}

\begin{axiom}[Super-exponential Decay -- A3]
\(|\mathrm{Reg}_2(f)(n)| \le C_f \exp(-c \log^2 n)\) for test functions with polynomial growth.
\end{axiom}

\begin{axiom}[Preservation of Standard Part -- A4]
Symmetric averaging commutes with the standard part map.
\end{axiom}

\begin{axiom}[Ultrafilter Independence -- A5]
Even quantities are independent of the choice of non-principal ultrafilter.
\end{axiom}

\section{Graded Cancellation Lemma}

\begin{lemma}[Exact Cancellation of Non-Invariant Modes]
Let \(\sigma: v \mapsto \pi - v\) be the reflection isometry of the right conoid. This induces a \(\mathbb{Z}_2\)-grading \(\mathcal{H} = \mathcal{H}_+ \oplus \mathcal{H}_-\).

Define the signed odd-sector measure \(\nu_-\) by equipping the positive spectral measure \(\mu_-\) with the sign induced by \(\sigma\). Then \(\nu_- \mapsto -\nu_-\) under the grading action.

Since \(\mathrm{Reg}_2\) extends linearly to signed measures and satisfies \(\mathrm{Reg}_2(-\nu) = -\mathrm{Reg}_2(\nu)\), and this antisymmetry is preserved under the ultrapower, it follows that
\[
\mathrm{Reg}_2(\nu_-) = 0.
\]
Therefore,
\[
K_{\rm Reg_2}(t)\Big|_{\mathcal{H}_-} = 0.
\]
This cancellation is algebraic and independent of eigenvalue interleaving.
\end{lemma}

\section{Poisson Summation Identity}

\begin{theorem}
The regularized heat kernel satisfies
\[
K_{\rm Reg_2}(t) = t^{-1} \Xi_{\rm Reg_2}(t) \, K_{\rm Reg_2}(1/t) + R_{\rm bridge}(t),
\]
where \(|R_{\rm bridge}(t)| \ll \exp\bigl(-c \log^2 (1/t)\bigr)\) for small \(t > 0\).
\end{theorem}

\begin{proof}
After the Cancellation Lemma, only even modes contribute. Decompose:
\[
K_{\rm Reg_2}(t) = K_{\rm cyl}(t) + K_{\rm curv}(t) + K_{\rm bridge}(t).
\]

\textbf{Cylindrical part:} Transforms exactly under \(t \to 1/t\).

\textbf{Curvature part:} Transforms with signs compatible with the prefactor \(\Xi_{\rm Reg_2}(t)\).

\textbf{Bridge part:} The non-modular remainder is suppressed by the super-exponential decay of the Dual-Zero regulator and finite support of the bridges, yielding the stated bound on \(R_{\rm bridge}(t)\).
\end{proof}

\section{Control of the Error Term \(E(s)\)}

\begin{theorem}[Strong Decay on the Critical Line]
For all \(|t| \ge 1\),
\[
|E(1/2 + it)| \le \exp(-c \log^2 |t|)
\]
for some explicit \(c > 0\).
\end{theorem}

\begin{theorem}[No Creation or Cancellation of Zeros for Large \(|t|\)]
There exists \(T_0 > 0\) such that for \(|t| > T_0\), any zero of \(\zeta_{\rm SAM3}(1/2 + it)\) must arise from the main term \(\Phi(1/2 + it) \cdot \zeta_{\rm SAM3}(1/2 - it)\).
\end{theorem}

\begin{proof}[Sketch]
From the functional equation, \(\Phi(1/2 + it) \cdot \zeta_{\rm SAM3}(1/2 - it) = -E(1/2 + it)\). The main term has a lower bound decaying slower than the error term. By reverse triangle inequality, \(E(s)\) cannot create or cancel zeros for large \(|t|\).
\end{proof}

\begin{theorem}[Explicit Bound in Compact Disk]
For any fixed \(R_0 > 0\), \(|E(s)| \le C(R_0, \omega_0)\) for all \(|s| \le R_0\).
\end{theorem}

\begin{corollary}
Combined with the large-\(|t|\) result, analytic control holds over the entire spectrum of \(\zeta_{\rm SAM3}(s)\).
\end{corollary}

\begin{proof}[Outline]
The Cancellation Lemma, Poisson identity, and error term all converge uniformly as \(\omega_0 \to 1^-\), recovering the geometric spectral zeta function of the conoid multiplied by the explicit factor \(G(s)\).
