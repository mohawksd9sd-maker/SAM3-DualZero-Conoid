\begin{abstract}
This document establishes the core definitions and key formulas for the SAM3-DualZero-Conoid framework. It includes the geometric base (infinite right conoid), the Dual-Zero hyperreal regularization, an explicit construction of the twelve binary-icosahedral bridges, and an explicit (proposed) construction of the spectral triple \((\mathcal{A}, \mathcal{H}, D)\) for the SAM3-DualZero-Conoid framework, including a precise statement of the axioms satisfied in both the Riemannian and Lorentzian cases. This document provides a precise axiomatic definition of the Dual-Zero regulator \(\mathrm{Reg}_2(\varepsilon)\) used in the SAM3-DualZero-Conoid framework. It isolates the minimal set of properties required for the spectral action computations, heat kernel expansion, and phenomenological predictions. Both derived and claimed relations are presented, with gaps noted for further development.
\end{abstract}

\section{Right Conoid Geometry}

\subsection{Definition and Metric}

The infinite right conoid is parametrized by
\[
\mathbf{r}(u,v) = \bigl( v \cos u,\ v \sin u,\ h(u) \bigr).
\]

For the recommended working model (right helicoid, \( h(u) = c u \)):
\begin{equation}
ds^2 = (v^2 + c^2)\, du^2 + dv^2.
\end{equation}

The infinite right conoid (working model: right helicoid) has metric
\begin{equation}
ds^2 = (v^2 + c^2)\, du^2 + dv^2.
\end{equation}

\subsection{Curvature}

Gaussian curvature:
\begin{equation}
K = -\frac{c^2}{(v^2 + c^2)^2} < 0.
\end{equation}

Mean curvature: \( H = 0 \) (minimal surface).

\begin{equation}
K = -\frac{c^2}{(v^2 + c^2)^2} < 0, \qquad H = 0.
\end{equation}

The metric on the right helicoid is
\begin{equation}
ds^2 = (v^2 + c^2)\, du^2 + dv^2,
\end{equation}
with Gaussian curvature \( K = -\frac{c^2}{(v^2 + c^2)^2} < 0 \).

\subsection{Orthonormal Coframe and Volume Form}

\[
e^1 = \sqrt{v^2 + c^2}\, du, \quad e^2 = dv,
\qquad
\mathrm{vol} = \sqrt{v^2 + c^2}\, du \wedge dv.
\]

\subsection{Dirac Operator (Schematic / Geometric Part)}

The Dirac operator on the right helicoid takes the form
\[
D = i \gamma^1 \left( \frac{1}{\sqrt{v^2 + c^2}} \partial_u + \cdots \right) 
  + i \gamma^2 \partial_v + \text{connection terms}.
\]

The geometric Dirac operator on the helicoid is
\[
D_{\text{geo}} = i \gamma^1 \left( \frac{1}{\sqrt{v^2 + c^2}} \partial_u + \cdots \right) 
  + i \gamma^2 \partial_v + \text{spin connection terms}.
\]

\begin{remark}
Due to the non-compact nature of the surface and linear growth of the area element along the rulings, the spectrum of \(D\) is continuous. Dual-Zero hyperreal regularization is required to obtain a well-defined spectral action.
\end{remark}

\section{Dual-Zero Hyperreal Structure}

\subsection{Definition}

\begin{definition}[Dual-Zero Hyperreal Structure]
A Dual-Zero structure on the hyperreals consists of a pair of signed infinitesimals \(\varepsilon, \delta\) equipped with a duality involution \(*\) satisfying \(\varepsilon^* = \delta\) and \(\delta^* = \varepsilon\).
\end{definition}

Let \(*\mathbb{R}\) be a hyperreal extension of the real numbers \(\mathbb{R}\). A \textbf{Dual-Zero structure} on \(*\mathbb{R}\) consists of a distinguished pair of signed infinitesimals (or dual monads around zero) together with a duality operation satisfying the following properties:
\begin{enumerate}
    \item There exist nonzero infinitesimals \(\varepsilon, \delta \in {}^*\mathbb{R}\) with \(\varepsilon \approx 0\) and \(\delta \approx 0\);
    \item A canonical duality map \((\cdot)^*\) acts on the infinitesimal neighborhood of zero such that \(\varepsilon^* = \delta\) and \(\delta^* = \varepsilon\);
    \item The pair generates a symmetric infinitesimal ideal compatible with the ordered field operations of \(*\mathbb{R}\), allowing controlled corrections that survive after taking the standard part \(\mathrm{st}(\cdot)\).
\end{enumerate}

We denote the resulting number system by \({}^*\mathbb{R}_{\text{DZ}}\) (Dual-Zero hyperreals) when the above structure is imposed. The duality map \(*\) is an involution (\(*^2 = \mathrm{id}\)) compatible with the transfer principle.

\begin{remark}
The signed nature (\(\varepsilon > 0\), \(\delta < 0\)) is crucial for generating chiral and generational structures when combined with binary-icosahedral action.
\end{remark}

\subsection{Proposed Scaling with Parameters}

\begin{equation}
\varepsilon = \ell_0 \exp(-1/\omega_0), \qquad 
\delta = -\ell_0 \exp(-\omega_0).
\end{equation}

Let \(\omega_0 > 0\) be a fixed positive real parameter (the regulator amplitude). Define the sequence
\[
\varepsilon(n) := \omega_0 (-1)^n n^{-n}, \qquad n \in \mathbb{N}.
\]
Let \(\mathcal{U}\) be a non-principal ultrafilter on \(\mathbb{N}\). The associated hyperreal is the equivalence class
\[
\varepsilon := [\varepsilon(n)]_{\mathcal{U}} \in {}^*\mathbb{R}.
\]

The \textbf{Dual-Zero regulator} is the composite object
\[
\mathrm{Reg}_2(\varepsilon) := \mathrm{st}_{\mathcal{U}} \Bigl( \frac{\varepsilon(2n) + \varepsilon(2n+1)}{2} \Bigr),
\]
where \(\mathrm{st}_{\mathcal{U}}\) denotes the standard-part map with respect to \(\mathcal{U}\). We view \(\mathrm{Reg}_2(\varepsilon)\) as a positive infinitesimal that acts on functions (or measures) of the spectrum via the rule
\[
\mathrm{Reg}_2(f)(n) := \frac{f(2n) + f(2n+1)}{2}.
\]

\subsection{Claimed Physical Relations}

G_N = \frac{64\pi \ell_0^2}{45}.

\subsection{Axiomatic Properties of the Dual-Zero Regulator}

We isolate the properties that are actually used in the SAM3 constructions:

\begin{axiom}[Linearity -- A1]
\(\mathrm{Reg}_2\) extends by linearity to the vector space of finite signed measures on \([0,\infty)\):
\[
\mathrm{Reg}_2(a\mu_1 + b\mu_2) = a\,\mathrm{Reg}_2(\mu_1) + b\,\mathrm{Reg}_2(\mu_2).
\]
\end{axiom}

\begin{axiom}[Grading compatibility / Antisymmetry -- A2]
If \(\nu\) is a signed measure that is antisymmetric under a \(\mathbb{Z}_2\)-grading (i.e., \(\nu \mapsto -\nu\)), then
\[
\mathrm{Reg}_2(\nu) = 0.
\]
In particular, this holds for the signed odd-sector measure \(\nu_-\) induced by the conoid reflection \(\sigma\).
\end{axiom}

\begin{axiom}[Super-exponential decay -- A3]
For any test function \(f\) with at most polynomial growth,
\[
\bigl|\mathrm{Reg}_2(f)(n)\bigr| \le C_f \cdot \exp(-c \log^2 n)
\]
for some constants \(C_f, c > 0\) that depend on \(\omega_0\) and the growth of \(f\).
\end{axiom}

\begin{axiom}[Preservation of standard part -- A4]
If a sequence of functions \(f_n\) converges uniformly on compact sets to a continuous function \(f\), then
\[
\mathrm{st}\bigl(\mathrm{Reg}_2(f_n)\bigr) = f(0)
\]
in the limit where the regulator scale tends to zero (i.e., as the effective cutoff goes to infinity).
\end{axiom}

\begin{axiom}[Ultrapower independence for symmetric quantities -- A5]
Any quantity that is symmetric under the pairwise averaging (in particular, all even functions of the spectrum) yields the same standard part for every choice of non-principal ultrafilter \(\mathcal{U}\).
\end{axiom}

\subsection{Analytic Consequences}

From the axioms above one immediately obtains:

\begin{itemize}
\item The regularized heat kernel \(K_{\rm Reg_2}(t)\) admits a well-defined asymptotic expansion as \(t \to 0^+\) whose coefficients differ from the classical Seeley--DeWitt coefficients by terms that are \(O(\omega_0)\) or smaller.

\item The regularized spectral zeta function 
\[
\zeta_{\rm SAM3}(s) = \mathrm{Tr}_{\rm Reg_2}(|D_\varepsilon|^{-s})
\]
is meromorphic in a half-plane and admits analytic continuation to the whole plane once the Poisson summation identity is established.

\item On the critical line, the error term \(E(s)\) satisfies the strong decay bound
\[
|E(1/2 + it)| \le \exp(-c \log^2 |t|)
\]
for large \(|t|\), which follows directly from A3.
\end{itemize}

\subsection{Interaction with the Spectral Triple}

The regulator enters the Dirac operator as

D_\varepsilon = D_c + \mathrm{Reg}_2(\varepsilon) \cdot \mathbf{1} + \gamma_5 \otimes D_F.
\]
Because of A2, the added term \(\mathrm{Reg}_2(\varepsilon) \cdot \mathbf{1}\) respects the grading induced by the conoid reflection. Consequently, the Cancellation Lemma holds and only even modes survive in the regularized trace.

\section{Binary-Icosahedral Bridges}

\subsection{Definition of the Binary Icosahedral Group}

\begin{definition}[Binary Icosahedral Group]
The \textbf{binary icosahedral group} \(2I\) is the central extension
\[
1 \to \{\pm 1\} \to 2I \to A_5 \to 1
\]
of order 120, realized as a discrete subgroup of \(\mathrm{SU}(2)\).
\end{definition}

\subsection{Definition of a Bridge}

\begin{definition}[Binary-Icosahedral Bridge]
A binary-icosahedral bridge is a structure morphism that incorporates representations of \(2I\) into the spectral triple on the right conoid, weighted by the Dual-Zero infinitesimals. The model uses exactly **twelve** such bridges.
\end{definition}

\subsection{Proposed Explicit Construction of the Twelve Bridges}

We propose the following explicit construction:

Let the twelve bridges be located at equally spaced angles
\[
\theta_k = \frac{k \pi}{6}, \quad k = 0,1,\dots,11.
\]

For each bridge \(k\), define
\begin{equation}
B_k = \varepsilon \cdot \rho_k(g) + \delta \cdot \rho_k(g)^*,
\end{equation}
where \(\rho_k\) is a 2-dimensional representation of \(2I\) assigned to the \(k\)-th sector, and \(g \in 2I\) is chosen according to the conjugacy class associated with angle \(\theta_k\).

The total bridge contribution to the Dirac operator is
\begin{equation}
D_{\text{bridge}} = \sum_{k=0}^{11} B_k \otimes \chi_k(v),
\end{equation}
where \(\chi_k(v)\) is a localized cutoff function (e.g., Gaussian) supported near the ruling at angle \(\theta_k\).

This construction distributes the twelve bridges symmetrically around the axis of the conoid and weights each by the Dual-Zero pair \((\varepsilon, \delta)\).

Twelve bridges are placed at angles \(\theta_k = k\pi/6\), \(k = 0,\dots,11\).

Each bridge contributes
\begin{equation}
B_k = \varepsilon \cdot \rho_k(g) + \delta \cdot \rho_k(g)^*.
\end{equation}

\begin{remark}
This is a proposed explicit construction. The following details still require precise definition:
\begin{itemize}
    \item Assignment of specific representations \(\rho_k\) to each bridge.
    \item Explicit form of the cutoff functions \(\chi_k(v)\).
    \item Verification that \(D_{\text{bridge}}\) preserves self-adjointness and compatibility with the real structure.
\end{itemize}
\end{remark}

\section{Explicit Construction of the Spectral Triple \((\mathcal{A}, \mathcal{H}, D)\)}

\subsection{Algebra \(\mathcal{A}\)}

We propose the algebra
\[
\mathcal{A} = C^\infty_{\text{DZ}}(M) \otimes \mathbb{C}[2I] \otimes M_n(\mathbb{C}),
\]
where \(C^\infty_{\text{DZ}}(M)\) denotes smooth functions on the right conoid with Dual-Zero hyperreal coefficients, and \(\mathbb{C}[2I]\) encodes the bridge data.

\subsection{Hilbert Space \(\mathcal{H}\)}

Let \(\mathcal{H}\) be the completion of sections of the spinor bundle over the right conoid, tensored with the regular representation of \(2I\) and with coefficients in the Dual-Zero hyperreals:
\[
\mathcal{H} = L^2(M, S) \otimes L^2(2I) \otimes {}^*\mathbb{C}_{\text{DZ}}.
\]

For the **Lorentzian version**, we equip \(\mathcal{H}\) with an indefinite inner product (Krein space structure) compatible with a Lorentzian signature on an appropriate 3+1 dimensional lift or on a causal completion of the conoid geometry.

\subsection{Dirac Operator \(D\) (Lorentzian Version)}

We propose the following explicit form of the Dirac operator:
\begin{equation}
D = D_{\text{geo}} + D_{\text{bridge}} + D_{\text{DZ}},
\end{equation}
where:
- \(D_{\text{geo}}\) is the geometric Dirac operator coming from the right conoid (with Lorentzian signature imposed via Wick rotation or direct indefinite metric construction),
- \(D_{\text{bridge}} = \sum_{k=0}^{11} B_k \otimes \chi_k(v)\) incorporates the twelve binary-icosahedral bridges,
- \(D_{\text{DZ}}\) contains Dual-Zero corrections (regularization terms involving \(\varepsilon\) and \(\delta\)).

For the **Lorentzian spectral triple**, we require:
- An indefinite inner product on \(\mathcal{H}\) (Krein space),
- A fundamental symmetry operator \(\beta\) (with \(\beta^2 = 1\)) that defines the Lorentzian adjoint,
- Compatibility conditions between \(D\), the real structure \(J\), and the grading \(\gamma\).

\begin{remark}
A fully rigorous Lorentzian spectral triple in the sense of noncommutative geometry is still an active research area. The construction above is a proposed framework that combines the geometric Dirac operator on the conoid with bridge and Dual-Zero terms while adopting a Krein-space formulation to accommodate Lorentzian signature.
\end{remark}

\subsection{Real Structure and KO-Dimension}

We propose that the real structure \(J\) is constructed by combining complex conjugation on the conoid spinors with the standard real structure on the finite part coming from representations of \(2I\). The target KO-dimension is to be determined so that it is compatible with a realistic Standard Model sector after fluctuation.

\subsection{Axioms of the Spectral Triple}

We state the axioms satisfied by the proposed spectral triple, distinguishing the Riemannian and Lorentzian cases.

\subsubsection{Riemannian Case (Standard Connes Axioms)}

The following axioms are satisfied (or proposed to be satisfied) in the Riemannian setting:

\begin{enumerate}
    \item \textbf{\(\mathcal{A}\) is a unital *-algebra} represented faithfully on \(\mathcal{H}\).
    
    \item \textbf{\(D\) is densely defined and self-adjoint} on \(\mathcal{H}\), with compact resolvent \((D - \lambda)^{-1}\) for \(\lambda \notin \mathbb{R}\).
    
    \item \textbf{First-order condition}: For all \(a \in \mathcal{A}\),
    \[
    [[D, a], b] = 0 \quad \text{for all } b \in \mathcal{A}.
    \]
    
    \item \textbf{Bounded commutators}: \([D, a]\) is a bounded operator for every \(a \in \mathcal{A}\).
    
    \item \textbf{Even spectral triple}: There exists a grading operator \(\gamma\) (with \(\gamma^2 = 1\), \(\gamma^* = \gamma\)) that commutes with \(\mathcal{A}\) and anticommutes with \(D\).
    
    \item \textbf{Real structure}: There exists an anti-unitary operator \(J\) (real structure) satisfying:
    \begin{itemize}
        \item \(J^2 = \pm 1\) (depending on KO-dimension),
        \item \(J D = \pm D J\),
        \item \(J a J^{-1} = a^*\) for \(a \in \mathcal{A}\),
        \item Compatibility with the grading \(\gamma\).
    \end{itemize}
\end{enumerate}

\subsubsection{Lorentzian Case (Modified Axioms)}

For the Lorentzian version we propose the following modifications:

\begin{enumerate}
    \item \textbf{Krein space structure}: The space \(\mathcal{H}\) is equipped with an indefinite inner product \(\langle \cdot, \cdot \rangle\) of signature \((\infty, \infty)\). There exists a fundamental symmetry operator \(\beta\) (\(\beta^2 = 1\), \(\beta^* = \beta\)) such that \(\langle \psi, \phi \rangle = (\beta \psi | \phi)\) defines a positive definite inner product.
    
    \item \textbf{\(\beta\)-self-adjointness of \(D\)}: Instead of ordinary self-adjointness, we require
    \[
    D^\dagger = \beta D \beta = D
    \]
    (i.e., \(D\) is self-adjoint with respect to the Krein inner product).
    
    \item \textbf{Compact resolvent}: The resolvent \((D - \lambda)^{-1}\) is compact with respect to the positive definite inner product induced by \(\beta\), for \(\lambda\) outside the spectrum.
    
    \item \textbf{Modified first-order condition}: The commutator condition \([[D, a], b] = 0\) is preserved, but all adjoints and commutators are understood with respect to the Krein adjoint.
    
    \item \textbf{Real structure in Krein space}: The real structure \(J\) must satisfy compatibility conditions with the fundamental symmetry \(\beta\), typically of the form
    \[
    J \beta = \pm \beta J
    \]
    (sign depending on the chosen KO-dimension in Lorentzian signature).
    
    \item \textbf{Grading}: The grading operator \(\gamma\) continues to anticommute with \(D\) and commutes with \(\mathcal{A}\), but must also be compatible with \(\beta\).
\end{enumerate}

\begin{remark}
The Lorentzian axioms above follow the general framework of Lorentzian noncommutative geometry (Krein-space spectral triples). Full verification that the proposed \(D = D_{\text{geo}} + D_{\text{bridge}} + D_{\text{DZ}}\) satisfies all modified axioms (especially \(\beta\)-self-adjointness and the real structure compatibility) remains to be carried out in detail.
\end{remark}
